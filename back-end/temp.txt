              </div>
            </el-col>
          </el-row>
        </el-col>
        <el-col :span="8" style="display: flex; gap: 10px; border-left: 1px solid #EBEEF5; padding-left: 20px;">
          <el-button type="primary" icon="Filter" @click="applyGlobalFilters">统一提取全部文件</el-button>
          <el-dropdown @command="handleExportCommand">
            <el-button type="warning" plain>

            </el-button>
            <template #dropdown>
              <el-dropdown-menu>
                <el-dropdown-item command="sheet">多sheet表格下载 (.xlsx)</el-dropdown-item>
                <el-dropdown-item command="zip">zip单表格下载 (.zip)</el-dropdown-item>
              </el-dropdown-menu>
            </template>
          </el-dropdown>
        </el-col>
      </el-row>

    <el-card shadow="always" class="tabs-card" style="min-height: 500px">

      </div>
      
      <el-tabs

        closable 
        @tab-remove="removeTab"
      >
        <el-tab-pane 
          v-for="item in tabs" 
          :key="item.name" 
          :label="item.title" 
          :name="item.name"
        >
          <!-- 阈值筛选操作区 -->
          <div class="filter-bar mb-20 p-15 bg-gray">
            <el-row :gutter="20" align="middle" style="width: 100%;">
              <!-- 左侧 2x2 网格条件筛选 -->
              <el-col :span="16">
                <el-row :gutter="20">
                  <el-col :span="12" style="margin-bottom: 12px;">
                    <div class="flex-center">
                      <span class="filter-label" style="width: 60px; font-weight: bold; color: #606266;">DEPTH</span>
                      <el-input v-model="item.filters.DEPTH.min" placeholder="最小值" clearable style="width: 120px;" />
                      <span class="separator" style="margin: 0 10px; color: #909399;">-</span>
                      <el-input v-model="item.filters.DEPTH.max" placeholder="最大值" clearable style="width: 120px;" />
                    </div>
                  </el-col>
                  <el-col :span="12" style="margin-bottom: 12px;">
                    <div class="flex-center">
                      <span class="filter-label" style="width: 60px; font-weight: bold; color: #606266;">AC</span>
                      <el-input v-model="item.filters.AC.min" placeholder="最小值" clearable style="width: 120px;" />

                      <el-input v-model="item.filters.DEN.max" placeholder="最大值" clearable style="width: 120px;" />
                    </div>
                  </el-col>
                  <el-col :span="12">
                    <div class="flex-center">
                      <span class="filter-label" style="width: 60px; font-weight: bold; color: #606266;">GR</span>
                      <el-input v-model="item.filters.GR.min" placeholder="最小值" clearable style="width: 120px;" />
                      <span class="separator" style="margin: 0 10px; color: #909399;">-</span>
                      <el-input v-model="item.filters.GR.max" placeholder="最大值" clearable style="width: 120px;" />
                    </div>
                  </el-col>
                </el-row>
              </el-col>
              
              <!-- 右侧 按钮组与文本提示 -->
              <el-col :span="8" style="display: flex; flex-direction: column; align-items: flex-end; justify-content: center; border-left: 1px solid #EBEEF5; padding-left: 20px;">
                <div style="margin-bottom: 12px; display: flex; gap: 10px;">
                  <el-button type="primary" icon="Filter" @click="applyFilters(item)">提取数据</el-button>
                  <el-button type="info" icon="Refresh" @click="clearFilters(item)" plain>清除筛选</el-button>
                  <el-button type="success" icon="Download" @click="exportToExcel(item)" plain>生成 Excel</el-button>
                </div>
                <div class="count-tip" style="font-size: 13px; color: #606266;">
                  当前提取出 <strong class="text-primary" style="font-size: 16px;">{{ item.displayData.length }}</strong> 行记录
                </div>
              </el-col>
            </el-row>
          </div>

          <!-- 第一版表格：前端截取前 300 行展示数据 (或使用过滤后的 displayData ) -->
          <div class="table-wrapper">
            <el-table 
              :data="item.displayData.slice(0, 300)" 
              border 
              stripe 
              style="width: 100%"
              height="550"
              v-loading="item.loading"
              element-loading-text="正在极速解析测井数据中..."
              :header-cell-style="{ background: '#f5f7fa', color: '#303133', fontWeight: 'bold' }"
            >
              <!-- 增加自增序号体现测点 -->
              <el-table-column type="index" label="测点序列" width="90" align="center" fixed />
              <el-table-column prop="DEPTH" label="深度 (DEPTH)" min-width="120" align="right" />
              <el-table-column prop="AC" label="声波时差 (AC)" min-width="120" align="right" />
              <el-table-column prop="DEN" label="密度 (DEN)" min-width="120" align="right" />

<script setup>
import { ref } from 'vue'
import { ElMessage } from 'element-plus'
import { Download, ArrowDown } from '@element-plus/icons-vue'
import { uploadTxtFile, scanServerPath, exportExcel, exportBatchExcel, exportBatchExcelZip } from '@/api/file'

const serverPath = ref('')
const activeTabName = ref('')
const tabs = ref([])
let tabIndex = 0

const globalFilters = ref({
  DEPTH: { min: '', max: '' },
  AC: { min: '', max: '' },
  DEN: { min: '', max: '' },
  GR: { min: '', max: '' }
})

const applyGlobalFilters = () => {
  if (tabs.value.length === 0) {
    ElMessage.warning('目前没有任何文件可以提取')
    return
  }
  tabs.value.forEach(tab => {
    tab.filters = JSON.parse(JSON.stringify(globalFilters.value))
    applyFilters(tab)
  })
  ElMessage.success(`已统一套用筛选条件至全部 ${tabs.value.length} 个文件！`)
}

const handleExportCommand = async (command) => {
  if (command === 'sheet') {
    await exportBatch()
  } else if (command === 'zip') {
    await exportBatchZip()
  }
}

const exportBatch = async () => {
  if (tabs.value.length === 0) {
    ElMessage.warning('当前没有任何数据可以导出')
    return
  }
  
  const payload = tabs.value.map(tab => ({
    title: tab.title,
    data: tab.displayData
  }))

  try {
    ElMessage.info('正在请求后端生成批量聚合 Excel (多Sheet)，请稍候...')
    const blob = await exportBatchExcel(payload)
    const url = window.URL.createObjectURL(new Blob([blob]))
    const link = document.createElement('a')
    link.style.display = 'none'
    link.href = url
    link.setAttribute('download', '一键批量测井数据报表.xlsx')
    document.body.appendChild(link)
    link.click()
    document.body.removeChild(link)
    window.URL.revokeObjectURL(url)
    
    ElMessage.success('多Sheet表格打包导出成功！')
  } catch (err) {
    console.error(err)
    ElMessage.error('打包导出失败，请检查网络设置或查看后台日志')
  }
}

const exportBatchZip = async () => {
  if (tabs.value.length === 0) {
    ElMessage.warning('当前没有任何数据可以导出')
    return
  }
  
  const payload = tabs.value.map(tab => ({
    title: tab.title,
    data: tab.displayData
  }))

  try {
    ElMessage.info('正在请求后端生成压缩包，请稍候...')
    const blob = await exportBatchExcelZip(payload)
    const url = window.URL.createObjectURL(new Blob([blob]))
    const link = document.createElement('a')
    link.style.display = 'none'
    link.href = url
    link.setAttribute('download', '测井数据包.zip')
    document.body.appendChild(link)
    link.click()
    document.body.removeChild(link)
    window.URL.revokeObjectURL(url)
    
    ElMessage.success('ZIP压缩包导出成功！')
  } catch (err) {
    console.error(err)
    ElMessage.error('ZIP导出失败，请检查网络设置或查看后台日志')
  }
}

const handleLocalUpload = async (file) => {
  if (file.status !== 'ready') return

  const fileName = file.name
  if (tabs.value.some(t => t.title === fileName)) {
    ElMessage.warning(`文件 ${fileName} 已经存在窗口中`)
    return
  }

  const formData = new FormData()
  formData.append('file', file.raw)

  const newTabName = `tab_${++tabIndex}`
  tabs.value.push({
    title: fileName,
    name: newTabName,
    data: [],
    displayData: [],
    filters: {
      DEPTH: { min: '', max: '' },
      AC: { min: '', max: '' },
      DEN: { min: '', max: '' },
      GR: { min: '', max: '' }
    },
    loading: true
  })
  activeTabName.value = newTabName

  try {
    const res = await uploadTxtFile(formData)
    const tabObj = tabs.value.find(t => t.name === newTabName)
    if (tabObj) {
      tabObj.data = res.data
      tabObj.displayData = [...res.data]
      tabObj.loading = false
      ElMessage.success(`${fileName} 上传与解析完成`)
    }
  } catch (error) {
    const tabObj = tabs.value.find(t => t.name === newTabName)
    if(tabObj) tabObj.loading = false
  }
}

const handleScanPath = async () => {
  if (!serverPath.value) {
    ElMessage.error('请输入需要扫描的本地路径')
    return
  }
  
  const loadingMsg = ElMessage.success({
    message: `正在向服务器请求扫描目录: ${serverPath.value}...`,
    duration: 0
  })

  try {
    const resList = await scanServerPath({ path: serverPath.value })
    loadingMsg.close()
    
    if (!resList || resList.length === 0) {
      ElMessage.warning('目标目录下未扫描到包含有效数据的 txt 文件')
      return
    }

    let addedCount = 0
    resList.forEach(fileObj => {
       if (!tabs.value.some(t => t.title === fileObj.title)) {
         const newTabName = `tab_${++tabIndex}`
         tabs.value.push({
           title: fileObj.title,
           name: newTabName,
           data: fileObj.data,
           displayData: [...fileObj.data],
           filters: {
             DEPTH: { min: '', max: '' },
             AC: { min: '', max: '' },
             DEN: { min: '', max: '' },
             GR: { min: '', max: '' }
           },
           loading: false
         })
         activeTabName.value = newTabName
         addedCount++
       }
    })

    if (addedCount > 0) {
      ElMessage.success(`扫描完成！成功载入 ${addedCount} 个测井文件`)
    } else {
      ElMessage.info(`扫描完毕，但该路径下的文件已全部在当前工作台中`)
    }
  } catch (err) {
    loadingMsg.close()
  }
}

const removeTab = (targetName) => {
  const tbs = tabs.value
  let activeName = activeTabName.value
  if (activeName === targetName) {
    tbs.forEach((tab, index) => {
      if (tab.name === targetName) {
        const nextTab = tbs[index + 1] || tbs[index - 1]
        if (nextTab) {
          activeName = nextTab.name
        }
      }
    })
  }
  activeTabName.value = activeName
  tabs.value = tbs.filter((tab) => tab.name !== targetName)
}

const applyFilters = (tabObj) => {
  const { filters, data } = tabObj
  let result = [...data]
  
  const cols = ['DEPTH', 'AC', 'DEN', 'GR']
  cols.forEach(col => {
    const { min, max } = filters[col]
    const hasMin = typeof min === 'string' && min.trim() !== ''
    const hasMax = typeof max === 'string' && max.trim() !== ''
    
    if (hasMin || hasMax) {
      const minVal = hasMin ? parseFloat(min) : -Infinity
      const maxVal = hasMax ? parseFloat(max) : Infinity
      
      result = result.filter(item => {
        const currentVal = parseFloat(item[col])
        if (isNaN(currentVal)) return false
        return currentVal >= minVal && currentVal <= maxVal
      })
    }
  })
  
  tabObj.displayData = result
  ElMessage.success(`过滤完成: 依据条件匹配出 ${result.length} 条记录`)
}

const clearFilters = (tabObj) => {
  ['DEPTH', 'AC', 'DEN', 'GR'].forEach(col => {
    tabObj.filters[col] = { min: '', max: '' }
  })
  tabObj.displayData = [...tabObj.data]
  ElMessage.success('筛选条件已重置，列表已还原')
}

const exportToExcel = async (tabObj) => {
  if (!tabObj.displayData || tabObj.displayData.length === 0) {
    ElMessage.warning('当前没有任何数据可以导出')
    return
  }
  
  try {
    ElMessage.info('正在请求后端生成归档 Excel，请稍候...')
    const blob = await exportExcel(tabObj.displayData)
    const url = window.URL.createObjectURL(new Blob([blob]))
    const link = document.createElement('a')
    link.style.display = 'none'
    link.href = url
    link.setAttribute('download', `${tabObj.title}_归档报表.xlsx`)
    document.body.appendChild(link)
    link.click()
    document.body.removeChild(link)
    window.URL.revokeObjectURL(url)
    
    ElMessage.success('成果表导出成功')
  } catch (err) {
    console.error(err)
    ElMessage.error('报表导出失败，请检查网络设置或查看后台日志')
  }
}
</script>
